        function switchTab(type) {
            const audioBtn = document.getElementById('tab-audio');
            const postersBtn = document.getElementById('tab-posters');
            const audioContent = document.getElementById('content-audio');
            const postersContent = document.getElementById('content-posters');

            if (type === 'audio') {
                audioBtn.classList.add('border-primary', 'text-primary');
                audioBtn.classList.remove('border-transparent', 'text-on-surface-variant');
                postersBtn.classList.remove('border-primary', 'text-primary');
                postersBtn.classList.add('border-transparent', 'text-on-surface-variant');
                
                audioContent.classList.remove('hidden');
                postersContent.classList.add('hidden');
            } else {
                postersBtn.classList.add('border-primary', 'text-primary');
                postersBtn.classList.remove('border-transparent', 'text-on-surface-variant');
                audioBtn.classList.remove('border-primary', 'text-primary');
                audioBtn.classList.add('border-transparent', 'text-on-surface-variant');
                
                postersContent.classList.remove('hidden');
                audioContent.classList.add('hidden');
